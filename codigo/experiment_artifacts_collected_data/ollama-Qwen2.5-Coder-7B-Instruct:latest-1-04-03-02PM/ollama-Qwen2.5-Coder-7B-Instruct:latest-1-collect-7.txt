@startuml "C4_UC_Conteudo"

!include https://raw.githubusercontent.com/plantuml-stdlib/C4-PlantUML/master/C4_Context.puml

title "Sistema de Gestão Financeira Pessoal - Contenido - Casos de Uso"

' Atores
actor "Cliente Individual" as user
actor "Financiamento de Empresas" as finance #999999
actor "Cobrança e Pagamentos" as payment #999999
actor "Investimentos em Ativos" as investment #999999

' Sistema e casos de uso
rectangle "Sistema de Gestão Financeira Pessoal" {
    usecase "UC01: Visualizar Resumo Financeiro" as UC01
    usecase "UC02: Gerenciar Conta Bancária" as UC02
    usecase "UC03: Solicitar Financiamento de Empresas" as UC03
    usecase "UC04: Realizar Pagamentos" as UC04
    usecase "UC05: Monitorar Investimentos em Ativos" as UC05
    usecase "UC06: Categorizar Transações Manuallymente" as UC06
    
    note right of UC01
      **Req. ID:** FR03, FR04
      **Descrição:** Permite ao usuário visualizar todas as transações 
      financeiras de suas contas conectadas, ordenadas 
      cronologicamente.
    end note
    
    usecase "UC07: Gerenciar Cartões de Crédito" as UC07
    usecase "UC08: Solicitar Empréstimos Pessoais" as UC08
    usecase "UC09: Realizar Pagamentos Internacionais" as UC09
    
    ' Relacionamentos entre casos de uso
    UC01 ..> UC02 : <<include>>
    UC01 ..> UC06 : <<extend>>
    
    UC03 ..> UC07 : <<include>>
    
    UC04 ..> UC08 : <<include>>
    UC09 ..> UC08 : <<extend>>
}

' Relacionamentos ator-caso de uso
user --> UC01
user --> UC02
user --> UC06

finance --> UC03
payment --> UC04
investment --> UC05

@enduml
